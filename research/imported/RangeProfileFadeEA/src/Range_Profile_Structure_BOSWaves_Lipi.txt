func addSwing(array<float> prices, array<int> bars, array<int> sides, float price, int bi, int side) {
    bool sameSide = false
    if array.length(prices) > 0 {
        sameSide := array.get(sides, 0) == side
    }
    if sameSide {
        float last = array.get(prices, 0)
        bool replace = side == 1 ? price >= last : price <= last
        if replace {
            array.set(prices, 0, price)
            array.set(bars, 0, bi)
        }
    } else {
        array.unshift(prices, price)
        array.unshift(bars, bi)
        array.unshift(sides, side)
    }
    while array.length(prices) > swingMemory {
        array.pop(prices)
        array.pop(bars)
        array.pop(sides)
    }
}

func findRange(array<float> prices, array<int> bars, array<int> sides, float atrValue) {
    bool bestFound = false
    float bestTop = 0.0
    float bestBottom = 0.0
    int bestLeft = 0
    int bestHighs = 0
    int bestLows = 0
    int n = array.length(prices)
    int minEvents = math.max(minTouches * 2, minRotations + 1)
    float tolerance = atrValue * clusterATR
    if n >= minEvents and atrValue > 0 {
        int count = minEvents
        while count <= n {
            float highSum = 0.0
            float lowSum = 0.0
            float highMin = 0.0
            float highMax = 0.0
            float lowMin = 0.0
            float lowMax = 0.0
            int highCount = 0
            int lowCount = 0
            int oldest = bar_index
            int newest = 0
            int i = 0
            while i < count {
                float p = array.get(prices, i)
                int bi = array.get(bars, i)
                int side = array.get(sides, i)
                oldest := math.min(oldest, bi)
                newest := math.max(newest, bi)
                if side == 1 {
                    highSum += p
                    highCount += 1
                    highMin := highCount == 1 ? p : math.min(highMin, p)
                    highMax := highCount == 1 ? p : math.max(highMax, p)
                } else {
                    lowSum += p
                    lowCount += 1
                    lowMin := lowCount == 1 ? p : math.min(lowMin, p)
                    lowMax := lowCount == 1 ? p : math.max(lowMax, p)
                }
                i += 1
            }
            if highCount >= minTouches and lowCount >= minTouches {
                float top = highSum / highCount
                float bottom = lowSum / lowCount
                float width = top - bottom
                bool stable = highMax - highMin <= tolerance * 2.0 and lowMax - lowMin <= tolerance * 2.0
                bool widthOK = width >= atrValue * minWidthATR and width <= atrValue * math.max(maxWidthATR, minWidthATR)
                bool spanOK = newest - oldest >= minRangeBars
                bool rotationsOK = count - 1 >= minRotations
                if top > bottom and stable and widthOK and spanOK and rotationsOK {
                    bestFound := true
                    bestTop := top
                    bestBottom := bottom
                    bestLeft := oldest
                    bestHighs := highCount
                    bestLows := lowCount
                }
            }
            count += 1
        }
    }
    return RangeFit.new(bestFound, bestTop, bestBottom, bestLeft, bestHighs, bestLows)
}

func resetProfile(array<float> profile) {
    array.clear(profile)
    int i = 0
    while i < profileBins {
        array.push(profile, 0.0)
        i += 1
    }
}

func addProfileBar(array<float> profile, float barLow, float barHigh, float barVolume, float bottom, float top) {
    float height = top - bottom
    if height > 0 and barVolume > 0 {
        if barHigh == barLow {
            if barHigh >= bottom and barHigh <= top {
                int idx = math.floor((barHigh - bottom) / height * profileBins)
                idx := math.max(0, math.min(profileBins - 1, idx))
                array.set(profile, idx, array.get(profile, idx) + barVolume)
            }
        } else {
            float candleSpan = math.max(barHigh - barLow, syminfo.mintick)
            int i = 0
            while i < profileBins {
                float binLow = bottom + height * i / profileBins
                float binHigh = bottom + height * (i + 1) / profileBins
                float overlap = math.max(0.0, math.min(barHigh, binHigh) - math.max(barLow, binLow))
                if overlap > 0 {
                    array.set(profile, i, array.get(profile, i) + barVolume * overlap / candleSpan)
                }
                i += 1
            }
        }
    }
}

func rebuildProfile(array<float> profile, int left, float top, float bottom) {
    resetProfile(profile)
    int stored = array.length(barHighs)
    int span = math.min(math.min(bar_index - left, 1500), stored - 1)
    int i = 0
    while i <= span {
        int idx = stored - 1 - i
        addProfileBar(profile, array.get(barLows, idx), array.get(barHighs, idx), array.get(barVols, idx), bottom, top)
        i += 1
    }
}

func newRange(int left, float top, float bottom) {
    box shell = box.new(left, top, bar_index, bottom, border_color=noColor, bgcolor=noColor)
    array<box> gradient = array.new<box>(0)
    array<box> profile = array.new<box>(0)
    int i = 0
    while i < gradientRows {
        array.push(gradient, box.new(left, top, bar_index, bottom, border_color=noColor, bgcolor=noColor))
        i += 1
    }
    int j = 0
    while j < profileBins {
        array.push(profile, box.new(left, top, left, bottom, border_color=noColor, bgcolor=noColor))
        j += 1
    }
    float mid = math.avg(top, bottom)
    line upper = line.new(left, top, bar_index, top, color=noColor, width=2)
    line lower = line.new(left, bottom, bar_index, bottom, color=noColor, width=2)
    line middle = line.new(left, mid, bar_index, mid, color=noColor, width=1, style=line.style_dotted)
    line poc = line.new(left, mid, bar_index, mid, color=noColor, width=2)
    return RangeDraw.new(shell, gradient, profile, upper, lower, middle, poc)
}

func renderProfile(RangeDraw d, array<float> profile, int left, int right, float top, float bottom, color css, bool completed) {
    float maxVol = 0.0
    int pocIndex = 0
    int pn = array.length(profile)
    int i = 0
    while i < pn {
        float v = array.get(profile, i)
        if v > maxVol {
            maxVol := v
            pocIndex := i
        }
        i += 1
    }
    int totalBars = math.max(right - left + 1, 1)
    int maxProfileBars = math.max(2, math.round(totalBars * profileWidth / 100.0))
    maxProfileBars := math.min(maxProfileBars, totalBars)
    float height = top - bottom
    array<box> bins = d.profile
    int bn = array.length(bins)
    int j = 0
    while j < bn {
        box b = array.get(bins, j)
        float vol = pn > j ? array.get(profile, j) : 0.0
        float ratio = maxVol > 0 ? vol / maxVol : 0.0
        float density = math.pow(ratio, 0.72)
        float binBottom = bottom + height * j / profileBins
        float binTop = bottom + height * (j + 1) / profileBins
        if showProfile and ratio > 0 {
            int widthBars = math.max(1, math.round(maxProfileBars * math.pow(ratio, 0.85)))
            int alpha = int(math.max(42.0, (completed ? 94.0 : 91.0) - density * (completed ? 30.0 : 42.0)))
            box.set_lefttop(b, right - widthBars + 1, binTop)
            box.set_rightbottom(b, right, binBottom)
            box.set_bgcolor(b, tr(css, alpha))
        } else {
            box.set_bgcolor(b, noColor)
        }
        j += 1
    }
    float pocPrice = bottom + height * (pocIndex + 0.5) / profileBins
    line.set_xy1(d.poc, left, pocPrice)
    line.set_xy2(d.poc, right, pocPrice)
    line.set_color(d.poc, showPOC and maxVol > 0 ? tr(chart.fg_color, completed ? 52 : 28) : noColor)
}

func renderRange(RangeDraw d, array<float> profile, int left, int right, float top, float bottom, int highTouches, int lowTouches, color resolvedCol, bool completed) {
    float height = math.max(top - bottom, syminfo.mintick)
    box.set_lefttop(d.shell, left, top)
    box.set_rightbottom(d.shell, right, bottom)
    box.set_border_color(d.shell, noColor)
    array<box> layers = d.gradient
    int rows = array.length(layers)
    if rows > 0 {
        float rowHeight = height / rows
        int i = 0
        while i < rows {
            float rowTop = top - rowHeight * i
            float rowBottom = rowTop - rowHeight
            float rowPos = (i + 0.5) / rows
            float distance = math.abs(rowPos - 0.5) * 2.0
            float strength = math.pow(distance, 1.55)
            color rowCol = rowPos < 0.5 ? bearCol : bullCol
            int alpha = int(99.0 - strength * (completed ? 17.0 : 38.0))
            box layer = array.get(layers, i)
            box.set_lefttop(layer, left, rowTop)
            box.set_rightbottom(layer, right, rowBottom)
            box.set_bgcolor(layer, tr(rowCol, alpha))
            i += 1
        }
    }
    int upperAlpha = int(math.max(4.0, 43.0 - highTouches * 8.0))
    int lowerAlpha = int(math.max(4.0, 43.0 - lowTouches * 8.0))
    line.set_xy1(d.upper, left, top)
    line.set_xy2(d.upper, right, top)
    line.set_color(d.upper, tr(bearCol, completed ? 34 : upperAlpha))
    line.set_xy1(d.lower, left, bottom)
    line.set_xy2(d.lower, right, bottom)
    line.set_color(d.lower, tr(bullCol, completed ? 34 : lowerAlpha))
    float mid = math.avg(top, bottom)
    line.set_xy1(d.middle, left, mid)
    line.set_xy2(d.middle, right, mid)
    line.set_color(d.middle, showMidline ? tr(chart.fg_color, 80) : noColor)
    renderProfile(d, profile, left, right, top, bottom, completed ? resolvedCol : rangeCol, completed)
}

static int rangeOpened = 0
static int highTouches = 0
static int lowTouches = 0
static RangeDraw activeDraw = RangeDraw.new()
static int profileLastBar = -1
static int breakUpCount = 0
static int breakDownCount = 0
static int cooldown = 0
static int deviationDir = 0
static float deviationExtreme = 0.0
static int deviationExtremeBar = 0

bool signalBreakUp = false
bool signalBreakDown = false
bool signalDeviationHigh = false
bool signalDeviationLow = false

if cooldown > 0 {
    cooldown -= 1
}

// New range detection
if not rangeActive and cooldown <= 0 and not na(atr) {
    RangeFit fit = findRange(swingPrice, swingBar, swingSide, atr)
    if fit.found {
        int candidateLeft = math.max(fit.left, bar_index - 1500)
        rangeActive := true
        rangeTop := fit.top
        rangeBottom := fit.bottom
        rangeLeft := candidateLeft
        rangeOpened := bar_index
        highTouches := fit.highTouches
        lowTouches := fit.lowTouches
        activeDraw := newRange(rangeLeft, rangeTop, rangeBottom)
        rebuildProfile(rangeProfile, rangeLeft, rangeTop, rangeBottom)
        profileLastBar := bar_index
        breakUpCount := 0
        breakDownCount := 0
        deviationDir := 0
        deviationExtreme := 0.0
        deviationExtremeBar := 0
    }
}

// Edge adaptation and profile update
if rangeActive {
    float tolerance = atr * clusterATR
    bool edgesChanged = false
    if hasPh and bar_index - swingLen >= rangeLeft and math.abs(ph - rangeTop) <= tolerance {
        rangeTop += (ph - rangeTop) * edgeAdapt
        highTouches += 1
        edgesChanged := true
    }
    if hasPl and bar_index - swingLen >= rangeLeft and math.abs(pl - rangeBottom) <= tolerance {
        rangeBottom += (pl - rangeBottom) * edgeAdapt
        lowTouches += 1
        edgesChanged := true
    }
    if edgesChanged {
        rebuildProfile(rangeProfile, rangeLeft, rangeTop, rangeBottom)
        profileLastBar := bar_index
    } else if bar_index > profileLastBar {
        addProfileBar(rangeProfile, low, high, nz(volume, 0.0), rangeBottom, rangeTop)
        profileLastBar := bar_index
    }
}

// Breakouts and deviations (closed bars only)
int breakDirection = 0
if rangeActive and barstate.isconfirmed {
    float buffer = atr * breakBuffer
    bool outsideUp = close > rangeTop + buffer
    bool outsideDown = close < rangeBottom - buffer
    breakUpCount := outsideUp ? breakUpCount + 1 : 0
    breakDownCount := outsideDown ? breakDownCount + 1 : 0
    if deviationDir == 0 {
        if outsideUp {
            deviationDir := 1
            deviationExtreme := high
            deviationExtremeBar := bar_index
        } else if outsideDown {
            deviationDir := -1
            deviationExtreme := low
            deviationExtremeBar := bar_index
        }
    } else {
        if deviationDir == 1 and high > deviationExtreme {
            deviationExtreme := high
            deviationExtremeBar := bar_index
        }
        if deviationDir == -1 and low < deviationExtreme {
            deviationExtreme := low
            deviationExtremeBar := bar_index
        }
        bool backInside = close <= rangeTop and close >= rangeBottom
        if backInside {
            if showDeviations {
                addDeviation(deviationExtremeBar, deviationExtreme, deviationDir)
            }
            signalDeviationHigh := deviationDir == 1
            signalDeviationLow := deviationDir == -1
            deviationDir := 0
            deviationExtreme := 0.0
            deviationExtremeBar := 0
        }
    }
    if breakUpCount >= breakCloses {
        breakDirection := 1
    }
    if breakDownCount >= breakCloses {
        breakDirection := -1
    }
    if maxRangeAge > 0 and bar_index - rangeOpened >= maxRangeAge {
        breakDirection := 2
    }
}

// Resolve the range
if rangeActive and breakDirection != 0 {
    color resolvedCol = breakDirection == 1 ? bullCol : breakDirection == -1 ? bearCol : rangeCol
    renderRange(activeDraw, rangeProfile, rangeLeft, bar_index, rangeTop, rangeBottom, highTouches, lowTouches, resolvedCol, true)
    if breakDirection == 1 or breakDirection == -1 {
        bool bullishBreak = breakDirection == 1
        float entry = bullishBreak ? rangeTop : rangeBottom
        if showBreakLabel {
            setBreakTag(activeDraw, label.new(bar_index, entry, bullishBreak ? "BREAKOUT  ↑" : "BREAKDOWN  ↓", yloc=yloc.price, style=label.style_label_left, color=tr(resolvedCol, 88), textcolor=chart.fg_color, size=size.small))
        }
        drawPosition(activeDraw, breakDirection, bar_index, rangeTop, rangeBottom)
        signalBreakUp := bullishBreak
        signalBreakDown := not bullishBreak
    }
    array.unshift(rangeHistory, activeDraw)
    while array.length(rangeHistory) > historyCount {
        RangeDraw old = array.pop(rangeHistory)
        deleteRange(old)
    }
    rangeActive := false
    array.clear(swingPrice)
    array.clear(swingBar)
    array.clear(swingSide)
    breakUpCount := 0
    breakDownCount := 0
    deviationDir := 0
    deviationExtreme := 0.0
    deviationExtremeBar := 0
    cooldown := cooldownBars
}

// Render the live range
if rangeActive {
    renderRange(activeDraw, rangeProfile, rangeLeft, bar_index, rangeTop, rangeBottom, highTouches, lowTouches, rangeCol, false)
}

// ---------------------------------------------------------------- Candle colors

color candleCol = rangeCol
bool paintNow = false
if paintCandles and rangeActive and rangeTop > rangeBottom {
    paintNow := true
    float pos = math.max(0.0, math.min(1.0, (close - rangeBottom) / (rangeTop - rangeBottom)))
    candleCol := pos <= 0.5 ? color.from_gradient(pos, 0.0, 0.5, bullCol, rangeCol) : color.from_gradient(pos, 0.5, 1.0, rangeCol, bearCol)
    candleCol := tr(candleCol, candleTransparency)
    if close > rangeTop {
        candleCol := bullCol
    } else if close < rangeBottom {
        candleCol := bearCol
    }
}

// ---------------------------------------------------------------- Output

barcolor(paintNow ? candleCol : nilv.c)
plot(rangeActive ? rangeTop : nilv.f, title="Active Range High", display=display.data_window)
plot(rangeActive ? rangeBottom : nilv.f, title="Active Range Low", display=display.data_window)

alertcondition(signalBreakUp, title="Range Breakout", message="Range Profile Structure broke upward | {{exchange}}:{{ticker}}")
alertcondition(signalBreakDown, title="Range Breakdown", message="Range Profile Structure broke downward | {{exchange}}:{{ticker}}")
alertcondition(signalDeviationHigh, title="Range High Deviation", message="Range Profile Structure deviated above the range and confirmed back inside | {{exchange}}:{{ticker}}")
alertcondition(signalDeviationLow, title="Range Low Deviation", message="Range Profile Structure deviated below the range and confirmed back inside | {{exchange}}:{{ticker}}")
